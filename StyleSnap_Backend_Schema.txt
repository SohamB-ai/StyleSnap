1. Overview
StyleSnap has no external backend server. All data lives in the browser. This document defines every data structure, storage schema, message payload type, and export format schema used by the extension — collectively referred to as the "backend schema."
Three gaps identified in the TRD are fully resolved in this document:
Export generation transformation logic — the exact functions that convert ExtractionResult into each export format
IndexedDB cascade delete / eviction logic — what happens to child records when a parent extraction is removed
ZIP packaging pipeline — exact folder structure, naming conventions, and file manifest

Tier 3 (ephemeral)
chrome.storage.session
Extraction checkpoints during long operations — cleared on browser close
2. Root Data Model — ExtractionResult
ExtractionResult is the central data model. It is produced by the content script, relayed through the service worker, stored in IndexedDB, and consumed by every export generator. All other schema types are nested within it or reference it by id.
interface ExtractionResult {  // ── Identity ──────────────────────────────────────────────  id:          string;          // UUID v4 — primary key in IndexedDB  url:         string;          // Full URL of the extracted page  origin:      string;          // e.g. "https://stripe.com"  title:       string;          // document.title at time of extraction  favicon:     string;          // data URI (PNG, ≤4KB) or empty string  timestamp:   number;          // Unix ms — Date.now() at extraction start  duration:    number;          // Extraction time in ms  version:     string;          // StyleSnap version that produced this — e.g. "1.0.0"  // ── Core payload ──────────────────────────────────────────  tokens:      DesignTokens;    // Full design token set (always present)  assets:      AssetManifest;   // Images, SVGs, icons, favicon (always present)  // ── V2 additions ──────────────────────────────────────────  layout?:     LayoutStructure; // Section-by-section page structure  components?: Component[];     // Detected UI components  // ── V3 addition ───────────────────────────────────────────  animations?: AnimationReport; // Detected animation/scroll libraries  // ── Metadata ──────────────────────────────────────────────  warnings:    string[];        // Non-fatal issues (e.g. cross-origin sheet skipped)  confidence:  number;          // 0–1 overall extraction confidence score  detectedFramework: FrameworkType;}type FrameworkType = "vanilla" | "tailwind" | "tailwind-v4" | "css-in-js"                   | "styled-components" | "emotion" | "unknown";

3.2  ColorToken
interface ColorToken {  id:            string;      // UUID v4 — stable ref across exports  value:         string;      // Normalized hex: "#4F46E5"  hex:           string;      // Same as value (explicit alias)  hsl:           string;      // "hsl(243 75% 59%)"  rgb:           string;      // "rgb(79, 70, 229)"  opacity:       number;      // 0–1 (1 = fully opaque)  frequency:     number;      // Count of DOM elements using this colour  contexts:      ColorContext[];  semanticGroup: SemanticGroup;  cssVarName?:   string;      // "--color-primary" if sourced from a CSS variable  suggestedName: string;      // Auto-named: "Electric Indigo", "Midnight Navy" etc.}type ColorContext = "background" | "text" | "border" | "fill" | "outline" | "shadow";type SemanticGroup = "primary" | "secondary" | "accent" | "neutral"                   | "semantic-success" | "semantic-warning" | "semantic-error"                   | "surface" | "unknown";
Color naming uses a curated lookup table of 2,000+ common design colour names matched by HSL proximity (±8° hue, ±15% sat/light). Falls back to "Colour {index}" if no match found within threshold.

9. IndexedDB Operations & Cascade Delete
This section patches TRD Gap #2: the exact eviction / cascade delete logic was not specced in the TRD. It is fully defined here.
9.1  Save Extraction (with History Eviction)
export async function saveExtraction(  result: ExtractionResult,  db: IDBPDatabase<StyleSnapDB>): Promise<void> {  // 1. Write to IndexedDB  await db.put("extractions", {    id: result.id, url: result.url,    origin: result.origin, timestamp: result.timestamp,    result,  });  // 2. Update chrome.storage.local history index  const storage = await chrome.storage.local.get("stylesnap_history");  const history: HistoryStore = storage.stylesnap_history ?? { entries: [] };  const newEntry: HistoryEntry = buildHistoryEntry(result);  history.entries.unshift(newEntry);  // Prepend (newest first)  // 3. EVICTION: if > 10 entries, remove oldest  if (history.entries.length > 10) {    const evicted = history.entries.splice(10);  // Remove from index 10 onwards    // 4. CASCADE DELETE: remove evicted records from IndexedDB    await Promise.all(evicted.map(entry => cascadeDelete(entry.id, db)));  }  await chrome.storage.local.set({ stylesnap_history: history });}
9.2  Cascade Delete
export async function cascadeDelete(  id: string,  db: IDBPDatabase<StyleSnapDB>): Promise<void> {  // Delete in a single transaction for atomicity  const tx = db.transaction(["extractions", "screenshots"], "readwrite");  await Promise.all([    tx.objectStore("extractions").delete(id),    tx.objectStore("screenshots").delete(id),  // No-op if no screenshots  ]);  await tx.done;}// Called from: history eviction, user-initiated delete, "Clear all history"export async function clearAllHistory(  db: IDBPDatabase<StyleSnapDB>): Promise<void> {  const tx = db.transaction(["extractions", "screenshots"], "readwrite");  await Promise.all([    tx.objectStore("extractions").clear(),    tx.objectStore("screenshots").clear(),  ]);  await tx.done;  await chrome.storage.local.set({ stylesnap_history: { entries: [] } });}
9.3  Load Extraction (History Tap)
export async function loadExtraction(  id: string,  db: IDBPDatabase<StyleSnapDB>): Promise<ExtractionResult | null> {  const record = await db.get("extractions", id);  return record?.result ?? null;}// Load all extractions for a given origin (for history grouping)export async function loadByOrigin(  origin: string,  db: IDBPDatabase<StyleSnapDB>): Promise<ExtractionRecord[]> {  return db.getAllFromIndex("extractions", "by-origin", origin);}

async function generateFullZIP(  result: ExtractionResult,  screenshots?: ScreenshotRecord): Promise<Blob> {  const zip = new JSZip();  // 1. Text files — generated synchronously  zip.file("DESIGN.md",         generateDesignMD(result));  zip.file("tokens.json",       generateTokensJSON(result));  zip.file("tailwind.config.js",generateTailwindConfig(result));  if (result.components) {    zip.file("components.md",   generateComponentsMD(result));    for (const tool of AI_TOOLS) {      zip.file(`master-prompt-${tool}.txt`, generateMasterPrompt(result, tool));    }  }  // 2. Assets — fetch and add  const assets = zip.folder("assets")!;  const images = assets.folder("images")!;  const svgs   = assets.folder("svgs")!;  const icons  = assets.folder("icons")!;  for (const [i, img] of result.assets.images.entries()) {    const ext  = mimeToExt(img.mimeType);    const slug = slugify(img.alt).slice(0, 30) || "untitled";    const data = await fetchAsBlob(img.src);  // fetch() in SW context    images.file(`img-${pad(i+1, 3)}-${slug}.${ext}`, data);  }  for (const [i, svg] of result.assets.svgs.entries()) {    const folder = svg.isIcon ? icons : svgs;    folder.file(`svg-${pad(i+1, 3)}.svg`, svg.svgContent);  }  if (result.assets.favicon?.dataUri) {    const b64 = result.assets.favicon.dataUri.split(",")[1];    assets.file("favicon.png", b64, { base64: true });  }  // 3. Screenshots — add blobs directly  if (screenshots) {    const shots = zip.folder("screenshots")!;    shots.file("full-page.png", screenshots.fullPage);    for (const [i, s] of screenshots.sections.entries()) {      shots.file(`section-${pad(i, 2)}-${s.label}.png`, s.blob);    }  }  return zip.generateAsync({ type: "blob", compression: "DEFLATE" });}
13. Schema Versioning & Migration
13.1  Version Strategy
Schema version is tracked independently in two places:
chrome.storage.local → Settings.schemaVersion (integer, increments on breaking changes)
IndexedDB → database version parameter in openDB() call (integer, triggers upgrade callback)
These two version integers are independent — a settings schema change does not require an IndexedDB version bump and vice versa.
